// Current implemented syntax. Semantic and backend support are documented in docs/SUPPORT.md.
program         -> topLevelDecl* ( setupDecl lateDecl* )? loopDecl lateDecl* EOF ;
libraryModule   -> ( importDecl | functionDecl | extensionDecl )* EOF ;
lateDecl        -> functionDecl | timerDecl | deviceDecl | importDecl | extensionDecl ;
topLevelDecl    -> defDecl | varDecl | lateDecl ;
functionDecl    -> "pub"? "noInterrupt"? "fn" IDENTIFIER "(" parameters? ")" ( ":" sourceType )? block ;
sourceType      -> TYPE | "&" "str" | arrayType | "&" "mut"? ( TYPE | arrayType | "[" TYPE "]" ) ;
arrayType       -> "[" TYPE ";" INTEGER "]" ;
parameters      -> parameter ( "," parameter )* ;
parameter       -> IDENTIFIER ":" sourceType | "device" IDENTIFIER ":" IDENTIFIER ;
importDecl      -> "import" "extension"? qualifiedName ";" ;
extensionDecl   -> "pub"? "extension" IDENTIFIER "for" IDENTIFIER "{" functionDecl* "}" ;
qualifiedName   -> IDENTIFIER ( "::" IDENTIFIER )* ;
// Import paths start with app:: or std::. Extension imports end with an extension name.
// Ordinary module import enables qualified calls, not extension method lookup.
// Static device parameters and extension receivers currently support UartTx only.
// Extension methods have implicit static self. No state, nested extensions or device copies.
declaration     -> varDecl | statement ;
varDecl         -> "var" IDENTIFIER ( ":" sourceType ( "=" expression )? | "=" expression ) ";" ;
defDecl         -> "def" ( alias | "{" alias+ "}" ) ;
alias           -> IDENTIFIER "=" pin_ref PINMODE ";" ;
pin_ref         -> PORT "." INTEGER ;
setupDecl       -> "setup" block ;
loopDecl        -> "loop" block ;
statement       -> exprStmt | returnStmt | forStmt | ifStmt | whileStmt | noInterruptStmt | block ;
noInterruptStmt -> "noInterrupt" block ;
returnStmt      -> "return" expression? ";" ;
exprStmt        -> expression ";" ;
forStmt         -> "for" IDENTIFIER "=" expression "to" expression block ;
ifStmt          -> "if" expression block ( "else" ( ifStmt | block ) )? ;
whileStmt       -> "while" expression block ;
block           -> "{" declaration* "}" ;

// Precedence: lowest to highest. Assignment is right-associative; binary operators are left-associative.
expression      -> assignment ;
assignment      -> IDENTIFIER "=" assignment | indexed "=" assignment | "*" unary "=" assignment | ternary ;
indexed         -> primary "[" expression "]" ;
ternary         -> logic_or ( "?" logic_or ":" logic_or )? ;
logic_or        -> logic_and ( "or" logic_and )* ;
logic_and       -> equality ( "and" equality )* ;
equality        -> comparison ( ( "!=" | "==" ) comparison )* ;
comparison      -> term ( ( ">" | ">=" | "<" | "<=" ) term )* ;
term            -> factor ( ( "-" | "+" ) factor )* ;
factor          -> unary ( ( "*" | "/" ) unary )* ;
unary           -> ( "!" | "-" | "*" ) unary | "&" "mut"? unary | call ;
call            -> qualifiedName "(" arguments? ")" | IDENTIFIER "." IDENTIFIER "(" arguments? ")" | primary "." "clone" "(" ")" | indexed | primary ;
arguments       -> expression ( "," expression )* ;
primary         -> INTEGER | FLOAT | STRING | BOOLEAN | PINSTATE | PINMODE
                 | IDENTIFIER | "(" expression ")" | arrayLiteral ;
arrayLiteral    -> "[" expression ";" INTEGER "]" | "[" expression ( "," expression )* ","? "]" ;
// N is 1..1024; scalar element types only. Array indexing/borrowing requires a named place.
// Array-reference and slice types are call arguments/parameters only; no stored or returned refs.
// &[T] / &mut [T] accept whole-array borrows and reborrows; len():u16 is runtime length.
// No subranges yet. Named value.clone() takes no arguments and requires Clone.
// &T supports integer/bool/PinState; &mut a[i] checks bounds before remaining call arguments.
// Element loans conflict with the entire root array. *r accesses a sized referent;
// *r=value needs mutable access; moving an owner out is forbidden; (*r).clone() copies the object.
// Clone on a non-named receiver requires a place (grouping, dereference, grouped element).
// References cannot be rebound/stored/returned; &&T and computed reference selection are unsupported.
// mut is contextual, not a globally reserved keyword. See docs/ARRAYS.md for bounds/borrowing.

// alias and not remain reserved, but have no implemented productions.
// Only named functions and methods of named devices/text references/arrays can be called; no chained calls,
// general computed receivers or method values (place Clone is the exception). Functions are top-level or extension members;
// hardware devices are declared only at application top level, never in libraries.
// No trailing commas in function parameters or call arguments.
// Signature types: integers, bool, PinState, fixed arrays; &str and scalar/array refs as parameters.
// &str results and globals are parsed but rejected until escaping-borrow analysis exists.
// Local &str requires an initializer. Assignment copies the view, not its bytes.
// Text methods: len():u16; byte_or(index:u16, fallback:u8):u8 (fallback outside bounds).
// STRING is UTF-8, not NUL-terminated. Escapes: \n \r \t \0 \\ and \".
// Other escapes and physical newlines inside STRING are rejected.
// Omitted result type means no value. See docs/TEXT.md for representation and limits.
// return is legal only inside fn. Every value-returning function must have an explicit
// return on every structural path; loops alone do not establish this guarantee.
// Global initializers cannot call functions. See docs/FUNCTIONS.md for the full contract.

timerDecl       -> "timer" IDENTIFIER "=" IDENTIFIER "{" timerField* "}" ;
timerField      -> "every" ":" duration "," | "tolerance" ":" ( "+" | "-" )? duration "," | "interrupt" block ;
duration        -> INTEGER ( "ns" | "us" | "ms" | "s" ) ;
// Require exactly one every and interrupt; at most one tolerance. Timer1 only.

deviceDecl      -> "device" IDENTIFIER "=" IDENTIFIER "{" deviceFields "}" ;
deviceFields    -> baudField modeField | modeField baudField ;
baudField       -> "baud" ":" INTEGER "," ;
modeField       -> "mode" ":" "TX" "," ;
// Exactly one baud and mode, in either order. The comma after each field is required;
// no semicolon follows the closing brace. baud, mode and TX are contextual identifiers.
// The hardware identifier currently resolves only to UART0. Its methods are
// try_write_byte(u8): bool, write_byte(u8): void and flush(): void.
// Device declarations may precede setup/loop, occur between them, or follow loop.
// Devices are static resources, not runtime values. Clock, baud, resource ownership
// and call effects are semantic checks; see docs/UART_API_PROPOSAL.md and docs/CALL_EFFECTS.md.
// Module visibility, specialization and std layout: docs/STD_EXTENSIONS_PROPOSAL.md.
// Value contracts: docs/VALUES.md. .clone() resolves Clone in the common registry.
// Scalar arrays clone their contents; shared views clone the view. User markers remain future work.
// Fixed arrays and temporary array references: docs/ARRAYS.md.
